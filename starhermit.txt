# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.29.0
engine.minor=31
engine.capabilities=platform,storage,clipboard

# Keyboard actions (players rebind keys and pads in the runtime's Options).
# The table is also played with the pointer or a finger: tap a highlighted
# card on the table or in the hand tray to play it, tap Draw to draw.
engine.group.prev_card=Table
control.prev_card=ArrowLeft | Previous card
control.next_card=ArrowRight | Next card
control.play_card=Enter+Space+NumpadEnter | Play selected card
control.draw=KeyD | Draw
control.hint=KeyH | Hint
control.undo=KeyU | Undo (practice)
control.pause=Escape | Pause / back
control.camera=KeyC | Reframe table

# Gamepad defaults (the browser build's): D-pad left/right choose a card,
# South plays it, West draws, North hints, East and Start pause / go back,
# Select undoes and the right stick press reframes the table.
engine.gamepad.prev_card=DpadLeft+LeftStickLeft
engine.gamepad.next_card=DpadRight+LeftStickRight
engine.gamepad.play_card=South
engine.gamepad.draw=West
engine.gamepad.hint=North
engine.gamepad.undo=Select
engine.gamepad.pause=East+Start
engine.gamepad.camera=RightStickPress
